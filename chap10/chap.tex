\chapter{重积分}
\section{二重积分}
\begin{theorem}
计算二重积分的一般步骤如下：
\begin{itemize}
    \item 确定积分区域 $D$ 的形状和边界．
    \item 将二重积分转化为迭代积分，选择合适的积分顺序（先对 $x$ 积分还是先对 $y$ 积分）．
    \item 确定内外层积分的积分限，根据积分区域 $D$ 的边界条件进行设置．
    \item 计算内层积分，得到一个关于外层变量的函数，然后计算外层积分．
\end{itemize}   
\end{theorem}
\begin{example}
    计算二重积分 $\displaystyle\iint_D xy \d \sigma$，其中 $D$ 是由 $y^2 = x$ 和 $y = x -2$ 围成的区域．
\end{example}
\begin{solution}
首先确定积分区域 $D$ 的边界．曲线 $y^2 = x$ 与 $y = x - 2$ 的交点可以通过解方程组
\[\begin{cases}
y^2 = x \\
y = x - 2
\end{cases}\qquad\Rightarrow\qquad \begin{cases}
    (1,-1) \\
    (4,2)
\end{cases}\]
图像如下：
\begin{center}
\begin{tikzpicture}[scale=0.8]
    \draw[->] (-0.5,0) -- (5,0) node[right] {$x$};
    \draw[->] (0,-2) -- (0,3) node[above] {$y$};
    \draw[domain=0:2.2, smooth, variable=\y, blue] plot ({\y^2}, {\y});
    \draw[domain=0:1.2, smooth, variable=\y, blue] plot ({\y^2}, {-\y});
    \draw[domain=0.5:4.5, smooth, variable=\x, red] plot ({\x}, {\x - 2});
    \node at (1,-1) [below right] {$(1,-1)$};
    \node at (4,2) [above right] {$(4,2)$};
\end{tikzpicture}
\end{center}
根据图像可以看出，如果先对 $y$ 积分，也就是从垂直$x$方向从下往上积分，那么积分区域 $D$ 势必被分成两部分，以$(1,-1)$为分界点．因此我们选择先对 $x$ 积分．对于每个固定的 $y$，$x$ 的积分限由两条曲线 $y^2 = x$ 和 $y = x - 2$ 决定．于是二重积分可以写成：
\[\iint_D xy \d \sigma = \int_{-1}^{2} \left( \int_{y^2}^{y+2} xy \d x \right) \d y \]
分别计算内层积分和外层积分：
\begin{align*}
\int_{-1}^{2} \left( \int_{y^2}^{y+2} xy \d x \right) \d y &= 
\int_{-1}^{2} y\d y\left[\frac{1}{2}x^2\right]_{y^2}^{y+2} \\
&=\frac{1}{2} \int_{-1}^{2} y\left( y^2 + 4y + 4 - y^4 \right) \d y \\
&=\frac{1}{2} \left[ -\frac{1}{6}y^6 + \frac{1}{4}y^4 + \frac{4}{3}y^3 +2y^2 \right]_{-1}^{2} \\
&=\frac{45}{8}
\end{align*}
\end{solution}

\section{三重积分}
\subsection{投影法}
\begin{theorem}
设 $D$ 是空间中的一个有界闭区域，$f(x,y,z)$ 在 $D$ 上连续．如果 $D$ 的投影区域 $D_{xy}$ 在 $xy$ 平面上是一个平面区域，并且对于每个 $(x,y) \in D_{xy}$，存在 $z_1(x,y)$ 和 $z_2(x,y)$ 使得 $(x,y,z) \in D$ 当且仅当 $z_1(x,y) \le z \le z_2(x,y)$，则三重积分 $\displaystyle\iiint_D f(x,y,z) \d V$ 可以表示为迭代积分：
\[\iiint_D f(x,y,z) \d V = \iint_{D_{xy}}\d x\d y\int_{z_1(x,y)}^{z_2(x,y)} f(x,y,z) \d z\]
\end{theorem}
\begin{example}
    计算积分 $\displaystyle\iiint_\Omega x \d V$，其中 $\Omega$ 为坐标面与平面 $x+2y+z=1$ 围成的区域．
\end{example}
\begin{solution}
首先确定积分区域 $\Omega$ 的边界．平面 $x+2y+z=1$ 与坐标面的交点可以通过解方程组
\[\begin{cases}
x + 2y + z = 1 \\
x = 0 \\
y = 0 \\
z = 0
\end{cases}\]
解得交点为 $(1,0,0)$、$(0,\frac{1}{2},0)$ 和 $(0,0,1)$．因此积分区域 $\Omega$ 是由坐标面和平面 $x+2y+z=1$ 围成的四面体．
于是有
\begin{align*}
    I &= \iiint_\Omega x \d V = \iint_{D_{xy}}x \d x\d y \int_0^{1-x-2y}  \d z = \iint_{D_{xy}} x(1-x-2y) \d x\d y \\
      &= \int_{0}^{1} x\d x \int_0^{\frac{1-x}{2}} (1-x-2y) \d y = \int_0^1 x\left[ (1-x)\cdot\frac{1-x}{2} - \left(\frac{1-x}{2}\right)^2\right] \d x\\
      &= \int_{0}^{1} x \cdot \frac{(1-x)^2}{4} \d x = \frac{1}{48}
\end{align*}
\end{solution}
\subsection{截面法}
\begin{theorem}
设 $\Omega$ 是空间中的一个有界闭区域，$f(x,y,z)$ 在 $\Omega$ 上连续．如果 $\Omega$ 在 $z$ 轴上的投影区间为 $[z_1, z_2]$，并且对于每个 $z \in [z_1, z_2]$，过点 $(0,0,z)$ 且垂直于 $z$ 轴的平面与 $\Omega$ 相交的平面区域为 $D_z$，则三重积分 $\displaystyle\iiint_\Omega f(x,y,z) \d V$ 可以表示为：
\[\iiint_\Omega f(x,y,z) \d V = \int_{z_1}^{z_2} \d z \iint_{D_z} f(x,y,z) \d x \d y\]
\end{theorem}
\begin{example}
    计算积分 $\displaystyle\iiint_\Omega z^2 \d V$，其中 $\Omega$ 是由椭球面 $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}+\dfrac{z^2}{c^2}=1$ 围成的区域．
\end{example}
\begin{solution}
    积分区域 $\Omega$ 的形状如图所示：

        \draw[->] (-2.5,0) -- (2.5,0) node[right] {$x$};
        \draw[->] (0,-2.5) -- (0,2.5) node[above] {$z$};
        \draw[->] (-1,-1) -- (1,1) node[right] {$y$};
        \draw[blue, thick] (0,0) circle [x radius=2, y radius=1.5];
        \draw[dashed] (-2,0) arc (180:0:2 and 0.5);
        \draw (2,0) arc (0:-180:2 and 0.5);
        \fill[gray!20, opacity=0.5] (0,1) ellipse (1.73 and 0.4);
        \draw[red, thick] (0,1) ellipse (1.73 and 0.4);
        \node at (1.8,1.2) [right, red] {$D_z$};
        \draw[<->] (2.2,0) -- (2.2,1) node[midway, right] {$z$};
    \end{tikzpicture}
    \end{center}
\[
        I = \int_{-c}^{c}z^2 \d z\iint_{D_z} \d x \d y
\]
注意到$D_z$为椭圆区域$\dfrac{x^2}{a^2(1-\frac{z^2}{c^2})}+\dfrac{y^2}{b^2(1-\frac{z^2}{c^2})}=1$，因此其面积为$\pi ab(1-\frac{z^2}{c^2})$．于是有
\[ I = \pi ab\int_{-c}^{c}(1-\frac{z^2}{c^2})z^2\d z = \frac{4}{15}\pi abc^3\]

\section{重积分的对称性}
\subsection{普通对称性（奇偶性）}
\begin{theorem}
    函数的对称性往往可以简化积分的计算，具体来说：
    \begin{itemize}
        \item 若区域 $D$ (或 $\Omega$) 关于坐标面（或坐标轴）对称，而被积函数 $f$ 关于该对称变量为奇函数，则积分为 0．
        \item 若区域 $D$ (或 $\Omega$) 关于坐标面（或坐标轴）对称，而被积函数 $f$ 关于该对称变量为偶函数，则积分等于对称部分上积分的两倍．
    \end{itemize}
\end{theorem}

\begin{example}
    计算积分 $\iint_D e^{-x^2-y^2}-x^3e^y \d x \d y$，其中 $D$ 是以原点为中心，半径为 $a$ 的圆盘．
\end{example}
\begin{solution}
    \sout{圆神!启动！}\\
    换用极坐标，其中被积表达式中的第二项是关于$x$的奇函数，且区域 $D$ 关于 $y$ 轴对称，因此在圆盘$D$上积分为零．因此有
\[I = \iint_D e^{-x^2-y^2} \d x \d y = \int_0^{2\pi} \d \theta\int_0^a e^{-r^2}\cdot r \d r  = \pi (1 - e^{-a^2})\]
\end{solution}

\subsection{轮换对称性}
\begin{theorem}[变量置换对称性]
    设积分区域 $\Omega \subset \mathbb{R}^n$ 为可求体积的集合．若存在变量的一置换 $\sigma \in S_n$，使得对于任意点 $\bm{x} = (x_1, \dots, x_n) \in \mathbb{R}^n$，满足 $\bm{x} \in \Omega$ 当且仅当 $\sigma(\bm{x}) = (x_{\sigma(1)}, \dots, x_{\sigma(n)}) \in \Omega$（即 $\Omega$ 在该置换下保持不变），则对于定义在 $\Omega$ 上的任何可积函数 $f$，有：
    \[ \int_{\Omega} f(x_1, \dots, x_n) \d V = \int_{\Omega} f(x_{\sigma(1)}, \dots, x_{\sigma(n)}) \d V \]
    特别地，若 $\Omega$ 在自变量的所有排列下均保持不变（完全轮换对称），则：
    \[ \int_{\Omega} f(x_1, \dots, x_n) \d V = \frac{1}{n!} \sum_{\sigma \in S_n} \int_{\Omega} f(x_{\sigma(1)}, \dots, x_{\sigma(n)}) \d V \]
    在三重积分的情形下，若 $\Omega$ 关于 $x, y, z$ 的轮换对称（如球面或立方体），则有：
    \[ \iiint_\Omega f(x,y,z) \d V = \iiint_\Omega f(y,z,x) \d V = \iiint_\Omega f(z,x,y) \d V \]
\end{theorem}

\begin{example}
    设区域 $D=\{(x,y) | x^2+y^2 \le 4,\; x,y \ge 0\}$，$f(x)$ 为$D$上正值连续函数，$a,b$为常数，求：
    \[I=\iint_D\frac{a\sqrt{f(x)}+b\sqrt{f(y)}}{\sqrt{f(x)}+\sqrt{f(y)}} \d \sigma \]
\end{example}
\begin{solution}
    注意到积分区域$D$关于$y=x$对称，且被积函数满足轮换对称，记$g(x,y)=\dfrac{a\sqrt{f(x)}+b\sqrt{f(y)}}{\sqrt{f(x)}+\sqrt{f(y)}}$，因此有
    \[ I = \frac{1}{2}\iint_D \left[ g(x,y) + g(y,x) \right] \d x \d y = \frac{1}{2}\iint_D (a+b) \d x \d y = \frac{a+b}{2}\pi\]
\end{solution}

\begin{example}
    计算积分$\displaystyle\iiint_\Omega z^2\d V$，其中$\Omega = \{(x,y,z) \mid x^2+y^2+z^2 \le 1\}$．
\end{example}
\begin{solution}
    由于积分区域$\Omega$关于 $x=y, y=z, z=x$ 面均对称，根据轮换对称性有
    \[ \iiint_\Omega x^2 \d V = \iiint_\Omega y^2 \d V = \iiint_\Omega z^2 \d V\]
    因此
    \[ I = \iiint_\Omega z^2\d V = \frac{1}{3}\iiint_\Omega (x^2+y^2+z^2) \d V\]
    利用球坐标计算（或直接利用体积结果）：
    \[ I = \frac{1}{3} \int_0^{2\pi} \d\phi \int_0^\pi \sin\theta \d\theta \int_0^1 r^2 \cdot r^2 \d r = \frac{1}{3} \cdot 4\pi \cdot \frac{1}{5} = \frac{4\pi}{15}\]
\end{solution}
\section{重积分的导数}
\begin{theorem}[雷诺传输定理（Reynolds Transport Theorem）]
    设 $\Omega(t)\subset \R^n$ 是一个随时间 $t$ 运动变化的区域（控制体），其边界为 $\partial\Omega(t)$，边界上某点处的\textbf{单位外法向量}为 $\bm{n}$，边界在该处的移动速度（边界速度）为 $\bm{v}_b$。对于定义在该区域上的标量场 $f(\bm{x}, t)$，其$n$重积分的导数为：
    \[ \frac{\d}{\d t} \int_{\Omega(t)} f(\bm{x}, t) \d V = \int_{\Omega(t)} \pdv{f}{t} \d V_n + \int_{\partial\Omega(t)} (\bm{v}_b \cdot \bm{n})f(\bm{x}, t)  \d S_{n-1} \]
\end{theorem}
\begin{corollary}[重积分求导的特殊情况]
    在实际应用中，常见的重积分求导问题通常具有特定的几何结构，可以利用以下方法简化计算：
    \begin{itemize}
        \item \textbf{积分区域固定}：如果积分区域 $\Omega$ 不随时间变化，即 $\Omega(t) = \Omega$，则边界速度 $\bm{v}_b = \bm{0}$，雷诺传输定理简化为：
        \[ \frac{\d}{\d t} \int_{\Omega} f(\bm{x}, t) \d V = \int_{\Omega} \pdv{f}{t} \d V \]
        \item \textbf{被积函数独立于时间}：如果被积函数 $f$ 不随时间变化，即 $f(\bm{x}, t) = f(\bm{x})$，则 $\pdv{f}{t} = 0$，雷诺传输定理简化为：
        \[ \frac{\d}{\d t} \int_{\Omega(t)} f(\bm{x}) \d V = \int_{\partial\Omega(t)} (\bm{v}_b \cdot \bm{n})f(\bm{x})  \d S \]
        \item \textbf{积分区域线性变化}：如果积分区域 $\Omega(t)$ 以某种线性方式随时间变化，例如 $\Omega(t) = \alpha(t)\Omega_0$，其中 $\alpha(t)$ 是一个标量函数，$\Omega_0$ 是一个固定区域，则可以通过适当的变量替换将问题转化为积分区域固定的情况，从而简化计算。
    \end{itemize}
    
\end{corollary}

\subsection{重积分变量替换公式（雅可比行列式）}
\begin{theorem}[变量替换定理]
    设 $\Omega'$ 是 $u_1, u_2, \dots, u_n$ 空间中的有界闭区域，变换 $\bm{x} = \bm{g}(\bm{u})$（即 $x_i = g_i(u_1, \dots, u_n)$）满足：
    \begin{itemize}
        \item $\bm{g}$ 在 $\Omega'$ 上具有连续的一阶偏导数；
        \item $\bm{g}$ 将 $\Omega'$ 一一对应地映射为 $\bm{x}$ 空间中的区域 $\Omega$；
        \item 在 $\Omega'$ 内，雅可比行列式 $J = \dfrac{\partial(x_1, \dots, x_n)}{\partial(u_1, \dots, u_n)} \neq 0$．
    \end{itemize}
    则对于 $\Omega$ 上的连续函数 $f(\bm{x})$，有：
    \[ \int_{\Omega} f(\bm{x}) \d V_x = \int_{\Omega'} f(\bm{g}(\bm{u})) |J| \d V_u \]
    其中雅可比行列式定义为：
    \[ J = \frac{\partial(x_1, \dots, x_n)}{\partial(u_1, \dots, u_n)} = \begin{vmatrix}
        \displaystyle\pdv{x_1}{u_1} & \displaystyle\pdv{x_1}{u_2} & \cdots & \displaystyle\pdv{x_1}{u_n} \\[1.5ex]
        \displaystyle\pdv{x_2}{u_1} & \displaystyle\pdv{x_2}{u_2} & \cdots & \displaystyle\pdv{x_2}{u_n} \\[1.0ex]
        \vdots & \vdots & \ddots & \vdots \\[0.5ex]
        \displaystyle\pdv{x_n}{u_1} & \displaystyle\pdv{x_n}{u_2} & \cdots & \displaystyle\pdv{x_n}{u_n}
    \end{vmatrix} \]
\end{theorem}

\begin{example}
    利用极坐标变换证明：$\displaystyle\iint_{\R^2} e^{-(x^2+y^2)} \d x \d y = \pi$．由此导出高斯积分 $\displaystyle\int_{-\infty}^{+\infty} e^{-x^2} \d x = \sqrt{\pi}$．
\end{example}
\begin{solution}
    考虑在圆盘 $D_R = \{(x,y) \mid x^2+y^2 \le R^2\}$ 上的积分．引入极坐标变换：
    \[ \begin{cases} x = r\cos\theta \\ y = r\sin\theta \end{cases} \implies J = \frac{\partial(x,y)}{\partial(r,\theta)} = \det \begin{pmatrix} \cos\theta & -r\sin\theta \\ \sin\theta & r\cos\theta \end{pmatrix} = r \]
    计算圆盘上的积分：
    \[ \iint_{D_R} e^{-(x^2+y^2)} \d x \d y = \int_0^{2\pi} \d\theta \int_0^R e^{-r^2} r \d r = 2\pi \left[ -\frac{1}{2}e^{-r^2} \right]_0^R = \pi(1-e^{-R^2}) \]
    令 $R \to +\infty$，得到整个平面的积分：
    \[ I = \iint_{\mathbb{R}^2} e^{-(x^2+y^2)} \d x \d y = \lim_{R \to +\infty} \pi(1-e^{-R^2}) = \pi \]
    另一方面，利用累次积分（ Fubini 定理）：
    \[ I = \int_{-\infty}^{+\infty} e^{-x^2} \d x \int_{-\infty}^{+\infty} e^{-y^2} \d y = \left( \int_{-\infty}^{+\infty} e^{-x^2} \d x \right)^2 \]
    从而证得 $\displaystyle\int_{-\infty}^{+\infty} e^{-x^2} \d x = \sqrt{\pi}$．
\end{solution}

\noindent \textbf{雅可比行列式的几何意义}：
\begin{itemize}
    \item \textbf{局部伸缩率}：$\abs{J}$ 描述了从 $\bm{u}$ 空间映射到 $\bm{x}$ 空间时，局部体积单元的\textbf{放大或缩小倍数}．
    \item \textbf{线性逼近}：在极小区域内，非线性变换 $\bm{g}$ 可以看作一个线性变换（由雅可比矩阵描述）．线性变换对体积的改变恰好是其变换矩阵的行列式．
\end{itemize}

\subsection{单位向量的微分与演化}
在正交曲线坐标系中，基向量随位置的变化可以通过从几何结构出发的解析方法严格推导。\sout{这里主要为物理服务，但其实都用不到}

\noindent \textbf{1. 严谨逻辑：从坐标曲线到单位向量} \\
设位置矢量为 $\bm{r} = \bm{r}(u_1, u_2, u_3)$。其几何构建逻辑如下：
\begin{enumerate}
    \item \textbf{坐标曲线}：固定其余坐标，仅让 $u_i$ 变化时形成的轨迹。
    \item \textbf{切矢量}：沿坐标曲线方向的矢量，定义为 $\bm{v}_i = \displaystyle\pdv{\bm{r}}{u_i}$。
    \item \textbf{度量因子（拉梅系数）}：切矢量的模长，记为 $h_i = \abs{\bm{v}_i} = \abs{\displaystyle\pdv{\bm{r}}{u_i}}$。
    \item \textbf{单位向量}：将切矢量归一化，即 $\bm{\hat{e}}_i = \displaystyle\frac{\bm{v}_i}{h_i} = \frac{1}{h_i} \pdv{\bm{r}}{u_i}$。
\end{enumerate}
对于正交坐标系，线元满足 $\d s^2 = \d \bm{r} \cdot \d \bm{r} = \sum h_i^2 \d u_i^2$。

\noindent \textbf{2. 核心定理：单位向量的导数公式} \\
在正交坐标系中，单位向量 $\bm{\hat{e}}_i$ 对坐标 $u_j$ 的偏导数满足：
\begin{theorem}
    \[ \frac{\partial \bm{\hat{e}}_i}{\partial u_j} = \begin{cases} 
        \displaystyle \frac{1}{h_i} \frac{\partial h_j}{\partial u_i} \bm{\hat{e}}_j & (i \neq j) \\[2ex]
        \displaystyle -\sum_{k \neq i} \frac{1}{h_k} \frac{\partial h_i}{\partial u_k} \bm{\hat{e}}_k & (i = j) 
    \end{cases} \]
\end{theorem}

\begin{proof}
    \noindent \textbf{(1) 情形一：$i \neq j$（交叉项）} \\
根据位置矢量 $\bm{r}$ 的二阶混合偏导数相等：
\[ \pdv{}{u_j} \left( \pdv{\bm{r}}{u_i} \right) = \pdv{}{u_i} \left( \pdv{\bm{r}}{u_j} \right) \]
代入切矢量与度量因子的关系 $\pdv{\bm{r}}{u_i} = h_i \bm{\hat{e}}_i$：
\[\pdv{}{u_i} (h_j \bm{\hat{e}}_j) \implies \pdv{h_i}{u_j} \bm{\hat{e}}_i + h_i \pdv{\bm{\hat{e}}_i}{u_j} = \pdv{h_j}{u_i} \bm{\hat{e}}_j + h_j \pdv{\bm{\hat{e}}_j}{u_i} \]
两边点乘 $\bm{\hat{e}}_j$，由正交性 $\bm{\hat{e}}_i \cdot \bm{\hat{e}}_j = 0$ ($i \neq j$) 及单位向量微分的性质（$\pdv{\bm{\hat{e}}_j}{u_i} \cdot \bm{\hat{e}}_j = 0$）：
\[ 0 + h_i \pdv{\bm{\hat{e}}_i}{u_j} \cdot \bm{\hat{e}}_j = \pdv{h_j}{u_i} \cdot 1 + 0 \implies \pdv{\bm{\hat{e}}_i}{u_j} \cdot \bm{\hat{e}}_j = \frac{1}{h_i} \pdv{h_j}{u_i} \]
可以发现在正交系中 $\pdv{\bm{\hat{e}}_i}{u_j}$ 仅在 $\bm{\hat{e}}_j$ 方向有分量，故：
\[ \pdv{\bm{\hat{e}}_i}{u_j} = \frac{1}{h_i} \pdv{h_j}{u_i} \bm{\hat{e}}_j \quad (i \neq j) \]

\noindent \textbf{(2) 情形二：$i = j$（自相关项）} \\
由于 $\bm{\hat{e}}_i$ 是单位向量，其对任意坐标的导数必垂直于自身，即 $\pdv{\bm{\hat{e}}_i}{u_i} \cdot \bm{\hat{e}}_i = 0$。因此 $\pdv{\bm{\hat{e}}_i}{u_i}$ 可表示为其他基向量的线性组合：
\[ \pdv{\bm{\hat{e}}_i}{u_i} = \sum_{k \neq i} \left( \pdv{\bm{\hat{e}}_i}{u_i} \cdot \bm{\hat{e}}_k \right) \bm{\hat{e}}_k \]
利用公式 $\bm{\hat{e}}_i \cdot \bm{\hat{e}}_k = 0 \implies \pdv{(\bm{\hat{e}}_i \cdot \bm{\hat{e}}_k)}{u_i} = 0$，得 $\pdv{\bm{\hat{e}}_i}{u_i} \cdot \bm{\hat{e}}_k = -\bm{\hat{e}}_i \cdot \pdv{\bm{\hat{e}}_k}{u_i}$。
由情形一已知 $\pdv{\bm{\hat{e}}_k}{u_i} = \frac{1}{h_k} \pdv{h_i}{u_k} \bm{\hat{e}}_i$，代入上式：
\[ \pdv{\bm{\hat{e}}_i}{u_i} \cdot \bm{\hat{e}}_k = -\bm{\hat{e}}_i \cdot \left( \frac{1}{h_k} \pdv{h_i}{u_k} \bm{\hat{e}}_i \right) = -\frac{1}{h_k} \pdv{h_i}{u_k} \]
代回求和式即得证：
\[ \pdv{\bm{\hat{e}}_i}{u_i} = -\sum_{k \neq i} \frac{1}{h_k} \pdv{h_i}{u_k} \bm{\hat{e}}_k \]
\end{proof}


\noindent \textbf{4. 极坐标系下的单位向量微分} \\
在极坐标 $(\rho, \phi)$ 中，$\d s^2 = \d \rho^2 + \rho^2 \d \phi^2$，故 $h_\rho = 1, h_\phi = \rho$。代入定理：
$\pdv{\bm{\hat{\rho}}}{\phi} = \frac{1}{h_\rho} \pdv{h_\phi}{\rho} \bm{\hat{\phi}} = \frac{1}{1} \cdot 1 \cdot \bm{\hat{\phi}} = \bm{\hat{\phi}}$；
$\pdv{\bm{\hat{\phi}}}{\phi} = -\frac{1}{h_\rho} \pdv{h_\phi}{\rho} \bm{\hat{\rho}} = -1 \cdot \bm{\hat{\rho}}$。
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|c|c|l|}
\hline
\textbf{算子} & \textbf{结果} & \textbf{几何直观/备注} \\ \hline
$\displaystyle\pdv{\bm{\hat{\rho}}}{\phi}$ & $\bm{\hat{\phi}}$ & $\bm{\hat{\rho}}$ 的转动速度为 $\bm{\hat{\phi}}$ \\ \hline
$\displaystyle\pdv{\bm{\hat{\phi}}}{\phi}$ & $-\bm{\hat{\rho}}$ & $\bm{\hat{\phi}}$ 的转动导致其指向中心 \\ \hline
$\displaystyle\pdv{\bm{\hat{\rho}}}{\rho}, \pdv{\bm{\hat{\phi}}}{\rho}$ & $\bm{0}$ & 单位向量方向与距离无关 \\ \hline
\end{tabular}
\end{center}

定义 $\bm{\hat{r}}, \bm{\hat{\theta}}, \bm{\hat{\phi}}$ 为单位方向向量．它们的导数展现了复杂的耦合关系：

\noindent \textbf{5. 位置向量的全微分与度规}\\
通过单位向量微分，位置向量 $\bm{r}$ 的全微分为：
\begin{itemize}

    \item \textbf{球坐标}：$\d\bm{r} = \d r \bm{\hat{r}} + r \d\theta \bm{\hat{\theta}} + r\sin\theta \d\phi \bm{\hat{\phi}}$
\end{itemize}
取模平方即可得到长度微元 $\d s^2 = \d\bm{r} \cdot \d\bm{r}$．

\subsection{二重积分}
\begin{exercise}
    设$f(x)$在$[0,1]$上连续，且$f(x)=x+\int_x^1f(y)f(y-x)\d y$，求$\int_0^1f(x)\d x$
\end{exercise}
\begin{solution}
    对原等式 $f(x)=x+\int_x^1f(y)f(y-x)\d y$ 在 $[0,1]$ 上积分：
    \[ \int_0^1 f(x) \d x = \int_0^1 x \d x + \int_0^1 \d x \int_x^1 f(y)f(y-x) \d y \]
    第一项为 $\frac{1}{2}$．考察第二项二重积分 $I = \int_0^1 \d x \int_x^1 f(y)f(y-x) \d y$．
    积分区域为 $D: \{ (x,y) \mid 0 \le x \le 1, x \le y \le 1 \}$．交换积分次序得：
    \[ I = \int_0^1 f(y) \d y \int_0^y f(y-x) \d x \]
    对内层积分做变量替换 $t = y-x$，则 $\d x = -\d t$，且当 $x=0$ 时 $t=y$，当 $x=y$ 时 $t=0$．
    \[ \int_0^y f(y-x) \d x = \int_0^y f(t) \d t \]
    故
    \[ I = \int_0^1 f(y) \left( \int_0^y f(t) \d t \right) \d y \]
    令 $F(y) = \int_0^y f(t) \d t$，则 $F'(y) = f(y)$，且 $F(0) = 0, F(1) = \int_0^1 f(x) \d x$．
    于是
    \[ I = \int_0^1 F'(y) F(y) \d y = \left[ \frac{1}{2} F^2(y) \right]_0^1 = \frac{1}{2} \left[ \int_0^1 f(x) \d x \right]^2 \]
    设 $S = \displaystyle\int_0^1 f(x) \d x$，代入原式得：
    \[ S = \frac{1}{2} + \frac{1}{2} S^2 \implies S^2 - 2S + 1 = 0 \implies (S-1)^2 = 0 \]
    解得 $S = 1$．
\end{solution}
\begin{exercise}
    计算极限 $\displaystyle\lim_{r \to 0} \frac{1}{\pi r^2} \iint_{x^2+y^2 \le r^2} e^{x^2-y^2} \cos (x+y) \d x \d y$．
\end{exercise}
\begin{solution}
    由积分中值定理知
    \[I=\lim_{r\to 0}\frac{e^{\xi^2-\eta^2}\cos(\xi+\eta)\cdot\pi r^2}{\pi r^2}=\lim_{\substack{\xi\to 0\\\eta\to 0}} e^{\xi^2-\eta^2}\cos(\xi+\eta) = 1 \]
    其中 $(\xi, \eta)\in D$．
\end{solution}

\subsection{三重积分}
\begin{exercise}
    计算积分 $\displaystyle\iiint_\Omega (x+y+z^3) \d V$，其中 $\Omega$ 是由曲面$x^2+y^2+z^2=2z(z>1)$与$z=\sqrt{x^2+y^2}$所围成的区域．
\end{exercise}
\begin{solution}
根据对称性，由于积分区域 $\Omega$ 关于 $x=0$ 和 $y=0$ 平面对称，且 $x, y$ 均为奇函数，因此 $\iiint_\Omega x \d V = 0$ 和 $\iiint_\Omega y \d V = 0$．只需要计算 $\iiint_\Omega z^3 \d V$．

采用球坐标系更为简便．曲面 $x^2+y^2+z^2=2z$ 即 $\rho = 2\cos\phi$．条件 $z>1$ 实际上对应于球面的上半部分，它与下方的圆锥面 $z=\sqrt{x^2+y^2}$ (即 $\phi = \frac{\pi}{4}$) 正好在 $z=1$ 处相交．因此区域 $\Omega$ 在球坐标下可表示为：
\[ \Omega = \{ (\rho, \theta, \phi) \mid 0 \le \theta \le 2\pi, \quad 0 \le \phi \le \frac{\pi}{4}, \quad 0 \le \rho \le 2\cos\phi \} \]
体积微元 $\d V = \rho^2\sin\phi \d\rho\d\phi\d\theta$．代入得：
\begin{align*}
\iiint_\Omega z^3 \d V &= \int_0^{2\pi} \d\theta \int_0^{\frac{\pi}{4}} \d\phi \int_0^{2\cos\phi} (\rho\cos\phi)^3 \rho^2\sin\phi \d\rho \\
&= 2\pi \int_0^{\frac{\pi}{4}} \cos^3\phi \sin\phi \left[ \frac{\rho^6}{6} \right]_0^{2\cos\phi} \d\phi \\
&= 2\pi \int_0^{\frac{\pi}{4}} \frac{32}{3}\cos^9\phi \sin\phi \d\phi \\
&= \frac{64\pi}{3} \left[ -\frac{\cos^{10}\phi}{10} \right]_0^{\frac{\pi}{4}} 
= \frac{32\pi}{15} \left( 1 - \frac{1}{32} \right) = \frac{31\pi}{15}
\end{align*}
\end{solution}

\begin{exercise}
    计算积分 $\displaystyle I = \iiint_\Omega y\cos(x+z)\d x\d y\d z$，其中 $\Omega$ 是由平面 $y=x,z=0,x+z=\frac{\pi}{2}$ 所围成的区域．
\end{exercise}
\begin{solution}
从围成区域的平面方程可知，隐含了 $y=0$（由 $y=x$ 及非负性可推出，否则区域无界；一般此类问题默认在第一卦限或 $y=0$ 为边界）。积分区域可表示为：
\[ \Omega = \left\{ (x,y,z) \;\bigg|\; 0 \le x \le \frac{\pi}{2}, \; 0 \le y \le x, \; 0 \le z \le \frac{\pi}{2}-x \right\} \]
化为累次积分计算：
\begin{align*}
I &= \int_0^{\frac{\pi}{2}} \d x \int_0^x y \d y \int_0^{\frac{\pi}{2}-x} \cos(x+z) \d z \\
  &= \int_0^{\frac{\pi}{2}} \left[ \frac{1}{2}y^2 \right]_0^x \left[ \sin(x+z) \right]_{z=0}^{z=\frac{\pi}{2}-x} \d x \\
  &= \frac{1}{2} \int_0^{\frac{\pi}{2}} x^2 \left( \sin\frac{\pi}{2} - \sin x \right) \d x \\
  &= \frac{1}{2} \int_0^{\frac{\pi}{2}} x^2 (1 - \sin x) \d x = \frac{1}{2} \int_0^{\frac{\pi}{2}} x^2 \d x - \frac{1}{2} \int_0^{\frac{\pi}{2}} x^2 \sin x \d x
\end{align*}
其中第一部分积分为：
\[ \frac{1}{2} \int_0^{\frac{\pi}{2}} x^2 \d x = \frac{1}{2} \left[ \frac{x^3}{3} \right]_0^{\frac{\pi}{2}} = \frac{\pi^3}{48} \]
第二部分积分使用分部积分法：
\begin{align*}
\int_0^{\frac{\pi}{2}} x^2 \sin x \d x &= \left[ -x^2 \cos x \right]_0^{\frac{\pi}{2}} + \int_0^{\frac{\pi}{2}} 2x \cos x \d x \\
&= 0 + 2 \left( [x \sin x]_0^{\frac{\pi}{2}} - \int_0^{\frac{\pi}{2}} \sin x \d x \right) \\
&= 2 \left( \frac{\pi}{2} - [-\cos x]_0^{\frac{\pi}{2}} \right) = \pi - 2
\end{align*}
综上所述：
\[ I = \frac{\pi^3}{48} - \frac{\pi - 2}{2} = \frac{\pi^3}{48} - \frac{\pi}{2} + 1 \]
\end{solution}

\begin{exercise}
    计算积分 $\displaystyle\iiint_\Omega (x+y+z)^2 \d V$，其中 $\Omega$ 是球体 $x^2+y^2+z^2 \le R^2$．
\end{exercise}
\begin{solution}
展开被积函数得 $(x+y+z)^2 = x^2+y^2+z^2 + 2xy + 2yz + 2zx$。
由于积分区域 $\Omega$ 关于原点及各个坐标平面对称，且交叉项 $xy, yz, zx$ 均为奇函数，因此它们在 $\Omega$ 上的积分均为 0，即：
\[ \iiint_\Omega 2xy \d V = \iiint_\Omega 2yz \d V = \iiint_\Omega 2zx \d V = 0 \]
从而原积分简化为：
\[ I = \iiint_\Omega (x^2+y^2+z^2) \d V \]
采用球坐标系计算，积分区域为 $0 \le \phi \le 2\pi$，$0 \le \theta \le \pi$，$0 \le r \le R$（这里 $\theta$ 为天顶角，$r$ 为极径），体积微元 $\d V = r^2\sin\theta \d r\d\theta\d\phi$：
\begin{align*}
I &= \int_0^{2\pi} \d\phi \int_0^\pi \sin\theta \d\theta \int_0^R r^2 \cdot r^2 \d r \\
  &= 2\pi \cdot \left[ -\cos\theta \right]_0^\pi \cdot \left[ \frac{r^5}{5} \right]_0^R \\
  &= 2\pi \cdot 2 \cdot \frac{R^5}{5} = \frac{4\pi R^5}{5}
\end{align*}
\end{solution}

\begin{exercise}
    计算积分 $\displaystyle\iiint_\Omega z^2 \d V$，其中 $\Omega :\{(x,y,z)\mid \begin{cases}
    x^2+y^2+z^2\le R^2\\
    x^2+y^2+(z-R)^2\le R^2
    \end{cases}\}$ 
\end{exercise}
\begin{solution}
区域 $\Omega$ 是由两个半径为 $R$ 的球体相交而成，分别以原点 $(0,0,0)$ 和 $(0,0,R)$ 为球心。
两个球面的交线满足 $z^2 = (z-R)^2$，解得交截面为 $z = \frac{R}{2}$。
因此，积分区域在 $z$ 方向上的范围为 $0 \le z \le R$。采用截面法（或者柱坐标系）来计算积分：
在高度 $z$ 处，平面的截面 $D_z$ 是一个圆：
\begin{itemize}
    \item 当 $0 \le z \le \frac{R}{2}$ 时，边界由下方的球面控制，即 $x^2+y^2 \le R^2 - (z-R)^2 = 2Rz - z^2$。
    \item 当 $\frac{R}{2} \le z \le R$ 时，边界由上方的球面控制，即 $x^2+y^2 \le R^2 - z^2$。
\end{itemize}
故在截面 $D_z$ 上的面积积分为 $\iint_{D_z} \d x\d y = \pi r_{max}^2$。于是原积分可以化为：
\begin{align*}
I = \iiint_\Omega z^2 \d V &= \int_0^{\frac{R}{2}} z^2 \cdot \pi(2Rz - z^2) \d z + \int_{\frac{R}{2}}^R z^2 \cdot \pi(R^2 - z^2) \d z \\
  &= \pi \int_0^{\frac{R}{2}} (2Rz^3 - z^4) \d z + \pi \int_{\frac{R}{2}}^R (R^2z^2 - z^4) \d z
\end{align*}
分别计算这两部分定积分：
第一部分：
\[ \pi \left[ \frac{Rz^4}{2} - \frac{z^5}{5} \right]_0^{\frac{R}{2}} = \pi \left( \frac{R^5}{32} - \frac{R^5}{160} \right) = \frac{\pi R^5}{40} = \frac{12\pi R^5}{480} \]
第二部分：
\begin{align*}
\pi \left[ \frac{R^2z^3}{3} - \frac{z^5}{5} \right]_{\frac{R}{2}}^R &= \pi \left( \frac{R^5}{3} - \frac{R^5}{5} \right) - \pi \left( \frac{R^5}{24} - \frac{R^5}{160} \right) \\
&= \pi \left( \frac{2}{15} - \frac{17}{480} \right) R^5 = \pi \left( \frac{64 - 17}{480} \right) R^5 = \frac{47\pi R^5}{480} 
\end{align*}
两者相加得到最终结果：
\[ I = \frac{12\pi R^5}{480} + \frac{47\pi R^5}{480} = \frac{59\pi R^5}{480} \]
\end{solution}

\begin{exercise}
    $\Sigma_1$是以点$(0,4,0)$为顶点且与曲面$\Sigma_2: \dfrac{x^2}{3}+\dfrac{y^2}{4}+\dfrac{z^2}{3}=1(y>0)$相切的圆锥面，求$\Sigma_1$与$\Sigma_2$所围成的区域的体积．
\end{exercise}
\begin{solution}
作变量代换：令 $u = \frac{x}{\sqrt{3}}, v = \frac{y}{2}, w = \frac{z}{\sqrt{3}}$，在此变换下，雅可比行列式 $J = \sqrt{3} \times 2 \times \sqrt{3} = 6$．

在 $uvw$ 空间中，$\Sigma_2$ 变为了单位上半球面 $u^2+v^2+w^2=1 (v>0)$，圆锥顶点变为了 $(0,2,0)$．
图形具有旋转对称性，可使用柱坐标系（以 $v$ 轴为中心轴，极径 $r = \sqrt{u^2+w^2}$）．
过顶点 $(0,2)$ 作单位圆 $r^2+v^2=1$ 的切线．设切线方程为 $r = k(2-v)$，代入圆方程得 $(k^2+1)v^2 - 4k^2v + 4k^2-1 = 0$．根据相切条件判别式 $\Delta=0$ 解得 $k = \frac{1}{\sqrt{3}}$，对应的切点高度为 $v = \frac{1}{2}$．

所以，在新空间中该区域的体积 $V'$ 可由一个圆锥的体积减去一个球冠的体积得到：
\begin{itemize}
    \item \textbf{圆锥体积}（顶点 $v=2$，底面在 $v=\frac{1}{2}$ 处，底面半径 $r=\frac{\sqrt{3}}{2}$）：
    \[ V_{cone} = \frac{1}{3}\pi \left(\frac{\sqrt{3}}{2}\right)^2 \times \left(2 - \frac{1}{2}\right) = \frac{3\pi}{8} \]
    \item \textbf{球冠体积}（$v$ 从 $\frac{1}{2}$ 到 $1$）：
    \[ V_{cap} = \int_{\frac{1}{2}}^1 \pi(1-v^2) \d v = \pi \left[ v - \frac{v^3}{3} \right]_{\frac{1}{2}}^1 = \frac{5\pi}{24} \]
\end{itemize}
使得 $V' = \frac{3\pi}{8} - \frac{5\pi}{24} = \frac{\pi}{6}$．
原体积 $V = \iiint_\Omega \d x\d y\d z = 6 \iiint_{\Omega'} \d u\d v\d w = 6 V' = \pi$．
\end{solution}

\begin{exercise}
    计算积分 $\displaystyle\iiint_\Omega \dfrac{xyz}{x^2+y^2}\d V$，其中 $\Omega$ 是曲面 $(x^2+y^2+z^2)^2=2a^2xy\ (a>0)$ 的第一象限部分． 
\end{exercise} 
\begin{solution}
采用球坐标系（取 $x=r\sin\theta\cos\phi, y=r\sin\theta\sin\phi, z=r\cos\theta$），第一象限对应 $0 \le \theta \le \frac{\pi}{2}$，且 $0 \le \phi \le \frac{\pi}{2}$．

代入曲面方程，得区域 $\Omega$ 的边界：
\[ r^4 = 2a^2(r\sin\theta\cos\phi)(r\sin\theta\sin\phi) \implies r^2 = a^2\sin^2\theta\sin(2\phi) \]
故极径的积分范围为 $0 \le r \le a\sin\theta\sqrt{\sin(2\phi)}$．

体积微元 $\d V = r^2\sin\theta \d r\d\theta\d\phi$．被积函数为：
\[ \frac{xyz}{x^2+y^2} = \frac{(r\sin\theta\cos\phi)(r\sin\theta\sin\phi)(r\cos\theta)}{r^2\sin^2\theta} = r\cos\theta\sin\phi\cos\phi \]
将积分转化为累次积分：
\begin{align*}
    I &= \int_0^{\frac{\pi}{2}} \d\phi \int_0^{\frac{\pi}{2}} \d\theta \int_0^{a\sin\theta\sqrt{\sin(2\phi)}} (r\cos\theta\sin\phi\cos\phi)(r^2\sin\theta) \d r \\
      &= \int_0^{\frac{\pi}{2}} \d\phi \int_0^{\frac{\pi}{2}} \sin\theta\cos\theta\sin\phi\cos\phi \left[ \frac{r^4}{4} \right]_0^{a\sin\theta\sqrt{\sin(2\phi)}} \d\theta \\
      &= \frac{a^4}{4} \int_0^{\frac{\pi}{2}} \d\phi \int_0^{\frac{\pi}{2}} \sin\theta\cos\theta\sin\phi\cos\phi \left( \sin^4\theta \cdot 4\sin^2\phi\cos^2\phi \right) \d\theta \\
      &= a^4 \int_0^{\frac{\pi}{2}} \sin^5\theta\cos\theta \d\theta \int_0^{\frac{\pi}{2}} \sin^3\phi\cos^3\phi \d\phi
\end{align*}
分别计算这两个单变量积分：
\[ \int_0^{\frac{\pi}{2}} \sin^5\theta\cos\theta \d\theta = \left[ \frac{1}{6}\sin^6\theta \right]_0^{\frac{\pi}{2}} = \frac{1}{6} \]
\[ \int_0^{\frac{\pi}{2}} \sin^3\phi\cos^3\phi \d\phi = \int_0^{\frac{\pi}{2}} \sin^3\phi (1-\sin^2\phi) \d(\sin\phi) = \frac{1}{4} - \frac{1}{6} = \frac{1}{12} \]
最终结果为：
\[ I = a^4 \times \frac{1}{6} \times \frac{1}{12} = \frac{a^4}{72} \]
\end{solution}

\begin{exercise}
    设 $\Omega$ 是由平面 $\Pi_1:x=0, \Pi_2:x=y, \Pi_3:y=z, \Pi_4:z=1, \Pi_5:z=2$ 所围成的空间区域，求三重积分 $\displaystyle\iiint_\Omega \frac{\d x\d y\d z}{\sqrt{y^2+z^2}}$．
\end{exercise}
\begin{solution}
由题目给出的各个平面方程式可以推断出，积分区域 $\Omega$ 的上下限关系为：
\[ 1 \le z \le 2, \quad 0 \le y \le z, \quad 0 \le x \le y \]
利用累次积分得：
\begin{align*}
    I &= \int_1^2 \d z \int_0^z \d y \int_0^y \frac{1}{\sqrt{y^2+z^2}} \d x \\
      &= \int_1^2 \d z \int_0^z \frac{y}{\sqrt{y^2+z^2}} \d y \\
      &= \int_1^2 \left[ \sqrt{y^2+z^2} \right]_{y=0}^{y=z} \d z \\
      &= \int_1^2 (\sqrt{2z^2} - \sqrt{z^2}) \d z \\
      &= \int_1^2 (\sqrt{2}-1)z \d z = (\sqrt{2}-1) \left[ \frac{z^2}{2} \right]_1^2 = \frac{3(\sqrt{2}-1)}{2}
\end{align*}
\end{solution}

\begin{exercise}
    设$f(x)$连续，$f(0)=a$，函数$F(t)=\displaystyle\iiint_\Omega [z+f(x^2+y^2+z^2)] \d V$，其中$\Omega :0\le z\le\sqrt{t^2-x^2-y^2}\,\text{及}\,z\ge\sqrt{x^2+y^2}$，求$\displaystyle\lim_{t\to 0}\frac{F(t)}{t^3}$．
\end{exercise}
\begin{solution}
区域 $\Omega(t)$ 由上半球面 $x^2+y^2+z^2 \le t^2$ ($z \ge 0$) 和圆锥面 $z \ge \sqrt{x^2+y^2}$ 围成。
采用球坐标系，积分区域可表示为：
\[ \Omega(t) = \left\{ (r, \theta, \phi) \;\bigg|\; 0 \le \theta \le 2\pi, \; 0 \le \phi \le \frac{\pi}{4}, \; 0 \le r \le t \right\} \]
首先计算区域 $\Omega(t)$ 的体积 $V(t)$（此时被积函数为 1）：
\begin{align*}
V(t) = \iiint_{\Omega(t)} \d V &= \int_0^{2\pi} \d\theta \int_0^{\frac{\pi}{4}} \sin\phi \d\phi \int_0^t r^2 \d r \\
&= 2\pi \cdot \left[ -\cos\phi \right]_0^{\frac{\pi}{4}} \cdot \left[ \frac{r^3}{3} \right]_0^t \\
&= 2\pi \left( 1 - \frac{\sqrt{2}}{2} \right) \frac{t^3}{3} = \frac{(2-\sqrt{2})\pi}{3} t^3
\end{align*}
由积分中值定理知，存在 $\Omega(t)$ 内的一点 $(\xi, \eta, \zeta)$，使得：
\[ F(t) = \iiint_{\Omega(t)} [z+f(x^2+y^2+z^2)] \d V = [\zeta + f(\xi^2+\eta^2+\zeta^2)] V(t) \]
当 $t \to 0$ 时，区域 $\Omega(t)$ 缩向原点 $(0,0,0)$，因此 $\zeta \to 0$，且 $(\xi^2+\eta^2+\zeta^2) \to 0$。由于 $f(x)$ 连续，且 $f(0)=a$，故：
\[ \lim_{t\to 0} [\zeta + f(\xi^2+\eta^2+\zeta^2)] = 0 + f(0) = a \]
于是，所求极限为：
\[ \lim_{t\to 0} \frac{F(t)}{t^3} = \lim_{t\to 0} \frac{[\zeta + f(\xi^2+\eta^2+\zeta^2)] V(t)}{t^3} = a \cdot \frac{V(t)}{t^3} = \frac{(2-\sqrt{2})\pi a}{3} \]
\end{solution}
\begin{myRemark}
    本题也可以采用洛必达法则来计算极限，这涉及重积分的求导。\\
    注意到积分区域线性变化，令 $x=tu, y=tv, z=tw$，由于区域 $\Omega(t)$ 变为固定的 $\Omega(1)$，且 $z \to tw, r \to t\rho$，则
    \[ F(t) = t^3 \iiint_{\Omega(1)} [tw + f(t^2(u^2+v^2+w^2))] \d u\d v\d w \]
    于是 $\displaystyle \lim_{t\to 0} \frac{F(t)}{t^3} = \lim_{t\to 0} \iiint_{\Omega(1)} [tw + f(t^2\rho^2)] \d u\d v\d w$。
    由 $f$ 的连续性，当 $t \to 0$ 时，被积函数趋于 $0 + f(0) = a$，直接得到 $a \iiint_{\Omega(1)} 1 \d u\d v\d w = a \cdot V(1) = \frac{(2-\sqrt{2})\pi a}{3}$。
\end{myRemark}

\subsection{重积分的应用}
\begin{exercise}
    求由曲线$y=\ln x$与两条直线$y=(e+1)-x$和$y=0$所围成的平面区域的面积．
\end{exercise}
\begin{solution}
首先求出各曲线之间的交点。
曲线 $y=\ln x$ 与直线 $y=0$ （$x$轴）的交点为 $(1, 0)$。
直线 $y=(e+1)-x$ 与 $y=0$ 的交点为 $(e+1, 0)$。
曲线 $y=\ln x$ 与直线 $y=(e+1)-x$ 的交点，经观察显然为 $x=e, y=1$（因为 $\ln e = 1$，且 $(e+1)-e = 1$）。

若将面积看作向 $y$ 轴投影进行积分（将 $x$ 表示为 $y$ 的函数），积分上下限为 $y=0$ 到 $y=1$。
同一高度 $y$ 处，左边界为 $x_1 = e^y$，右边界为 $x_2 = (e+1)-y$。
因此所求面积为：
\begin{align*}
S &= \int_0^1 \left( x_2 - x_1 \right) \d y \\
  &= \int_0^1 \left( (e+1)-y - e^y \right) \d y \\
  &= \left[ (e+1)y - \frac{1}{2}y^2 - e^y \right]_0^1 \\
  &= \left( (e+1) - \frac{1}{2} - e \right) - \left( 0 - 0 - 1 \right) \\
  &= \frac{1}{2} + 1 = \frac{3}{2}
\end{align*}
\end{solution}
\begin{exercise}
    求球面$x^2+y^2+z^2=4a^2$和柱面$x^2+y^2=2ax$所包围的且在柱面内部的体积．
\end{exercise}
\begin{solution}
积分区域由柱面 $x^2+y^2=2ax$ （即 $(x-a)^2+y^2=a^2$）和球面 $x^2+y^2+z^2=4a^2$ 组成。
由于立体关于 $xy$ 平面对称，我们可以先计算立体在第一卦限及第四卦限即 $z \ge 0$ 部分体积的两倍。
采用极坐标系：柱面方程化为 $r=2a\cos\theta$ ($-\frac{\pi}{2} \le \theta \le \frac{\pi}{2}$)，立体的上半部分曲面为 $z=\sqrt{4a^2-x^2-y^2} = \sqrt{4a^2-r^2}$。
因此体积为：
\begin{align*}
V &= 2 \iint_{D} \sqrt{4a^2-x^2-y^2} \d x \d y \\
  &= 2 \int_{-\frac{\pi}{2}}^{\frac{\pi}{2}} \d\theta \int_0^{2a\cos\theta} \sqrt{4a^2-r^2} \, r \d r \\
  &= 4 \int_0^{\frac{\pi}{2}} \left[ -\frac{1}{3}(4a^2-r^2)^{\frac{3}{2}} \right]_0^{2a\cos\theta} \d\theta \\
  &= -\frac{4}{3} \int_0^{\frac{\pi}{2}} \Big( [4a^2(1-\cos^2\theta)]^{\frac{3}{2}} - (4a^2)^{\frac{3}{2}} \Big) \d\theta \\
  &= \frac{32a^3}{3} \int_0^{\frac{\pi}{2}} \left( 1 - \sin^3\theta \right) \d\theta
\end{align*}
分别计算定积分：
\[ \int_0^{\frac{\pi}{2}} 1 \d\theta = \frac{\pi}{2} \]
\[ \int_0^{\frac{\pi}{2}} \sin^3\theta \d\theta = \int_0^{\frac{\pi}{2}} (1-\cos^2\theta)\sin\theta \d\theta = \left[ -\cos\theta + \frac{1}{3}\cos^3\theta \right]_0^{\frac{\pi}{2}} = \frac{2}{3} \]
代入上式得最终体积：
\[ V = \frac{32a^3}{3} \left( \frac{\pi}{2} - \frac{2}{3} \right) = \frac{16\pi a^3}{3} - \frac{64a^3}{9} \]
\end{solution}
\begin{exercise}
    设有一个半径为 $R$ 的球体，$P_0$ 是球面上的一个定点，球体内每个点 $P$ 的密度与该点到$P_0$的距离的平方成正比（比例常数$k>0$），求球体的重心位置．
\end{exercise}
\begin{solution}
为计算方便，我们建立空间直角坐标系，将球心放置在原点 $(0,0,0)$，并设球面上的定点 $P_0$ 为坐标 $(0,0,R)$。
球体区域为 $\Omega: x^2+y^2+z^2 \le R^2$。设球体内任意一点 $P(x,y,z)$，则其到 $P_0$ 的距离平方为 $d^2 = x^2+y^2+(z-R)^2 = x^2+y^2+z^2-2Rz+R^2$。
由题意，密度函数 $\rho(x,y,z) = k(x^2+y^2+z^2-2Rz+R^2)$。
由于球体及其密度分布关于 $z$ 轴对称，故重心的横、纵坐标 $\bar{x} = 0, \bar{y} = 0$。

我们计算球体的总质量 $M$ 和对 $xy$ 平面的静矩 $M_{xy}$。由于在中心对称区域上关于单数次幂变量如 $z$ 的积分为 0，有：
\begin{align*}
M &= \iiint_\Omega \rho(x,y,z) \d V = k \iiint_\Omega (x^2+y^2+z^2+R^2) \d V - 2kR \iiint_\Omega z \d V \\
  &= k \left( \iiint_\Omega (x^2+y^2+z^2) \d V + R^2 \iiint_\Omega 1 \d V \right) \\
  &= k \left( \int_0^R r^2 (4\pi r^2) \d r + R^2 \cdot \frac{4}{3}\pi R^3 \right) \\
  &= k \left( \frac{4\pi R^5}{5} + \frac{4\pi R^5}{3} \right) = \frac{32\pi k R^5}{15}
\end{align*}
接下来计算静矩 $M_{xy}$：
\begin{align*}
M_{xy} &= \iiint_\Omega z\rho(x,y,z) \d V = k \iiint_\Omega z(x^2+y^2+z^2+R^2-2Rz) \d V \\
       &= k \iiint_\Omega z(x^2+y^2+z^2+R^2) \d V - 2kR \iiint_\Omega z^2 \d V \\
       &= 0 - 2kR \iiint_\Omega z^2 \d V
\end{align*}
由对称性 $\iiint_\Omega z^2 \d V = \frac{1}{3} \iiint_\Omega (x^2+y^2+z^2) \d V = \frac{1}{3} \cdot \frac{4\pi R^5}{5} = \frac{4\pi R^5}{15}$。代入得：
\[ M_{xy} = -2kR \left( \frac{4\pi R^5}{15} \right) = -\frac{8\pi k R^6}{15} \]
因此，重心 $z$ 坐标为：
\[ \bar{z} = \frac{M_{xy}}{M} = \frac{-\frac{8\pi k R^6}{15}}{\frac{32\pi k R^5}{15}} = -\frac{R}{4} \]
综上所述，重心位置在过点 $P_0$ 与球心的直线上，位于球心偏向 $P_0$ 的反方向（另一侧），距球心的距离为 $\frac{R}{4}$。
\end{solution}

\subsection{技巧性问题}
\begin{exercise}
    设函数$f(x)$在区间$[0,1]$上连续，并设$\displaystyle\int_0^1 f(x) \d x = A$，求$\displaystyle\int_0^1 \d x \int_x^1 f(x)f(y) \d y$．
\end{exercise}
\begin{solution}
令 $I = \displaystyle\int_0^1 \d x \int_x^1 f(x)f(y) \d y$。
交换积分顺序，这相当于在正方形区域 $[0,1]\times[0,1]$ 的上半部分区域 $D_1 = \{(x,y) \mid 0 \le x \le 1, x \le y \le 1\}$ 上的二重积分：
\[ I = \iint_{D_1} f(x)f(y) \d x\d y \]
由对称性（将被积变量 $x$ 和 $y$ 互换），有：
\[ I = \iint_{D_1} f(y)f(x) \d y\d x = \int_0^1 \d y \int_0^y f(x)f(y) \d x\ \]
这相当于该函数在正方形区域的下半部分 $D_2 = \{(x,y) \mid 0 \le x \le 1, 0 \le y \le x\}$ 上的积分。
因此：
\[ 2I = \iint_{D_1\cup D_2} f(x)f(y) \d x\d y = \int_0^1 \d x \int_0^1 f(x)f(y) \d y \]
由于变量可以分离，故：
\[ 2I = \left(\int_0^1 f(x) \d x\right) \left(\int_0^1 f(y) \d y\right) = A \cdot A = A^2 \]
故 $I = \frac{A^2}{2}$。
\end{solution}

\begin{exercise}
    设函数$F(t)=\displaystyle\int_0^t\d x\int_0^x\d y\int_0^y f(z)\d z$，其中$f(z)$连续，试把$F(t)$化为对$z$的定积分，并求$F'''(t)$．
\end{exercise}
\begin{solution}
这是一道多次积分的问题，区域是由不等式 $0 \le z \le y \le x \le t$ 给出的四面体。
为了把 $F(t)$ 化为对 $z$ 的定积分，我们需要改变积分顺序，即把 $z$ 放到最外层积分。外层 $z$ 的取值范围为 $[0, t]$。在给定 $z$ 时，内侧变量 $x$ 和 $y$ 满足 $z \le y \le x \le t$：
\begin{align*}
F(t) &= \int_0^t f(z) \left( \int_z^t \d x \int_z^x \d y \right) \d z \\
     &= \int_0^t f(z) \left( \int_z^t (x - z) \d x \right) \d z \\
     &= \int_0^t f(z) \left[ \frac{(x-z)^2}{2} \right]_z^t \d z \\
     &= \int_0^t \frac{(t-z)^2}{2} f(z) \d z
\end{align*}
利用含参变量积分求导公式，分别求导：
\begin{align*}
F'(t) &= 0 + \int_0^t \frac{\partial}{\partial t}\left(\frac{(t-z)^2}{2}\right) f(z) \d z = \int_0^t (t-z)f(z) \d z \\
F''(t) &= 0 + \int_0^t \frac{\partial}{\partial t}\big((t-z)\big) f(z) \d z = \int_0^t f(z) \d z \\
F'''(t) &= f(t)
\end{align*}
\end{solution}

\begin{exercise}
    设函数$f(x)$连续且恒大于零，
    \[F(t)=\frac{\displaystyle \iiint_{\Omega(t)}f(x^2+y^2+z^2)\d V}{\displaystyle\iiint_{D(t)} f(x^2+y^2)\d \sigma},\quad G(t)=\frac{\displaystyle \iiint_{D(t)}f(x^2+y+2)\d \sigma}{\displaystyle\int_{-t}^tf(x^2)\d x},\]
    其中$\Omega(t)=\{(x,y,z)\mid x^2+y^2+z^2\le t^2\},\,D(t)=\{(x,y)\mid x^2+y^2\le t^2\}$．
    \begin{tasks}[label = (\arabic*)](2)
    \task 讨论$F(t)$在区间$(0,+\infty)$上的单调性；
    \task 证明当$t>0$时，$F(t)>\frac{2}{\pi}G(t)$．
    \end{tasks}
\end{exercise}
\begin{solution}
(1) 将 $F(t)$ 中的重积分用极坐标和球坐标表示：
\[ \iiint_{\Omega(t)} f(x^2+y^2+z^2) \d V = \int_0^{2\pi} \d\varphi \int_0^\pi \sin\theta \d\theta \int_0^t r^2 f(r^2) \d r = 4\pi \int_0^t r^2 f(r^2) \d r \]
\[ \iint_{D(t)} f(x^2+y^2) \d\sigma = \int_0^{2\pi} \d\varphi \int_0^t r f(r^2) \d r = 2\pi \int_0^t r f(r^2) \d r \]
故：
\[ F(t) = \frac{2\displaystyle\int_0^t r^2 f(r^2) \d r}{\displaystyle\int_0^t r f(r^2) \d r} \]
求导：
\[ F'(t) = 2\frac{\displaystyle t^2 f(t^2) \int_0^t f(r^2)r(t-r) \d r}{\displaystyle\left[ \int_0^t r f(r^2) \d r \right]^2} > 0\]
所以 $F(t)$ 在 $(0, +\infty)$ 上严格单调递增。

(2) 由于 $f$ 连续且恒正，化简 $G(t)$：
\[G(t) = \frac{\displaystyle 2\pi \int_0^t r f(r^2) \d r}{\displaystyle 2 \int_0^t f(x^2) \d x} = \pi \frac{\displaystyle\int_0^t r f(r^2) \d r}{\displaystyle\int_0^t f(r^2) \d r}\]
我们要证 $F(t) > \frac{2}{\pi} G(t)$，即：
\[\frac{\displaystyle\int_0^t r^2 f(r^2) \d r}{\displaystyle\int_0^t r f(r^2) \d r} > \frac{\displaystyle\int_0^t r f(r^2) \d r}{\displaystyle\int_0^t f(r^2) \d r} \]
注意到这等价于柯西-施瓦茨不等式的变体：
\[ \left( \int_0^t r f(r^2) \d r \right)^2 < \left( \int_0^t r^2 f(r^2) \d r \right) \left( \int_0^t f(r^2) \d r \right) \]
接下来我们就来证一遍柯西-施瓦茨不等式，令
\[g(t)=\int_0^tf(r^2)r^2\d r\int_0^tf(r^2)\d r-\left[\int_0^tf(r^2)r\d r\right]^2\]
求导：
\[g'(t)=f(t^2)\int_0^tf(r^2)(t-r)\d r > 0\]
因此 $g(t)$ 在 $(0, +\infty)$ 上单调递增，所以 $g(t) > g(0)=0$ 对 $t > 0$ 成立。
这就证明了结论。
\end{solution}
\begin{myRemark}
    柯西-施瓦茨不等式：
    \[\left[\int_a^bf(x)g(x)\d x\right]^2\le\int_a^bf^2(x)\d x\int_a^bg^2(x)\d x\]
    其中 $f(x), g(x)$ 是区间 $[a,b]$ 上的可积函数。当$f(x)$ 和 $g(x)$ 在$[a,b]$上几乎处处线性相关时等号成立。
\end{myRemark}
